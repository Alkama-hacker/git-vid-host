function gvh_nice($bytes)
{
    $u = array('B', 'KB', 'MB', 'GB', 'TB');
    $i = 0;
    $n = (float) $bytes;
    while ($n >= 1024 && $i < 4) {
        $n /= 1024;
        $i++;
    }
    return ($i === 0 ? sprintf('%d %s', $n, $u[$i]) : sprintf('%.1f %s', $n, $u[$i]));
}

function gvh_fail($msg)
{
    throw new RuntimeException($msg);
}

/** read + validate the container header */
function gvh_header($raw)
{
    global $MAGIC;
    $m = strlen($MAGIC);
    if (strlen($raw) < $m + 5 || substr($raw, 0, $m) !== $MAGIC) {
        gvh_fail('This is not a GVH vault file.');
    }
    $len = unpack('N', substr($raw, $m, 4));
    $len = $len[1];
    if ($len < 2 || $len > 65536 || strlen($raw) < $m + 4 + $len) {
        gvh_fail('The vault header is damaged.');
    }
    $json = substr($raw, $m + 4, $len);
    $header = json_decode($json, true);
    if (!is_array($header) || !isset($header['n'], $header['salt'], $header['iv'])) {
        gvh_fail('The vault header is not readable.');
    }
    return array($header, $json, $m + 4 + $len);
}

/**
 * Decrypt chunk $i. Returns the plaintext or throws.
 * IV = 8 random bytes from the header + uint32 BE chunk index
 * AAD = sha256(header json)
 */
function gvh_chunk($raw, $header, $hjson, $offset, $i, $key)
{
    $len = unpack('N', substr($raw, $offset, 4));
    $len = $len[1];
    $offset += 4;
    if ($len < 1 || $offset + $len + 16 > strlen($raw)) {
        gvh_fail('The vault file is truncated - re-upload it.');
    }
    $ct = substr($raw, $offset, $len);
    $tag = substr($raw, $offset + $len, 16);
    $pt = openssl_decrypt(
        $ct,
        'aes-256-gcm',
        $key,
        OPENSSL_RAW_DATA,
        base64_decode($header['iv']) . pack('N', $i),
        $tag,
        hash('sha256', $hjson, true)
    );
    if ($pt === false) {
        gvh_fail('Wrong password. Nothing was decrypted.');
    }
    return array($pt, $offset + $len + 16);
}
